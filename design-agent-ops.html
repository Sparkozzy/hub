Seção 1 · Contexto no Pipeline

O Lugar do Agente
de Design

O agente de design não é o primeiro nem o último no pipeline. Ele recebe output estruturado do agente de copywriting e entrega HTML renderizável para o Puppeteer. Entender este contexto é pré-requisito para o system prompt.

01
Strategy
Tema, ICP, objetivo, pilar editorial
→
02
Copywriter
Gera copy por slide: headline, corpo, dado, CTA
→
03 — AQUI
Design Agent
Recebe DesignBriefing, produz SlideHTML[]
→
04
Puppeteer
Renderiza HTML → PNG 1080×1350px
→
05
QA / Aprovação
Telegram gate para revisão humana

Responsabilidade do Agente de Design

Recebe: um DesignBriefing estruturado com o copy de cada slide já escrito pelo agente anterior.

Decide: qual layout, quais camadas, quais assets, quais tokens de cor e tipografia para cada slide — baseado nos documentos de especificação (DRS, VCS, Suplementos).

Produz: um array de SlideHTML — HTML+CSS completo por slide, pronto para o Puppeteer renderizar em 1080×1350px.

Não decide: o conteúdo do copy (vem do agente anterior), a aprovação final (vai para revisão humana), a publicação (outro agente downstream).

2.1 Input — DesignBriefing

// types/design-briefing.ts
// Gerado pelo agente de Copywriting, consumido pelo Design Agent

export type SlideType =
  | 'cover'        // Capa — LAYOUT-01
  | 'data'         // Dado numérico — LAYOUT-02
  | 'split-data'   // Dado + foto — LAYOUT-03
  | 'photo-full'   // Foto full-bleed — LAYOUT-04
  | 'chips'        // Lista de chips — LAYOUT-05
  | 'argument'     // Argumento editorial — LAYOUT-06
  | 'cta';         // CTA final — LAYOUT-07

export type AccentColor = 'blue' | 'mint' | 'amber';
export type BackgroundMode = 'dark' | 'light' | 'cta-blue';
export type PhotoTreatment = 'brand-tint' | 'editorial' | 'bw-blue' | 'vignette';

export interface SlideCopy {
  // Texto principal (âncora) — o elemento de maior escala
  anchor: {
    text: string;              // ex: "73%" ou "Sua equipe trabalha mais."
    accentText?: string;      // parte em cor de accent — ex: "receita não cresce."
    isNumber: boolean;        // true = tratar como número (DM Mono + escala hero)
    unit?: string;            // se isNumber: "%" | "min" | "R$" | etc
  };

  // Suporte da âncora (contexto imediato)
  support?: {
    text: string;            // ex: "das empresas mid-market perdem até 30%..."
    isBold: boolean;         // true = Syne Bold, false = DM Sans Regular
  };

  // Corpo de texto (zona SUPORTE, abaixo do separador)
  body?: string;             // ex: "Quando o fundador sai sem sistematizar..."

  // Chips — só para slideType 'chips'
  chips?: Array<{
    icon: string;           // "✓" | "→" | "⚡" — unicode simples, nunca emoji 3D
    text: string;           // max 5 palavras
  }>;

  // Dado com fonte — só para slideType 'data' e 'split-data'
  dataSource?: string;       // ex: "Estudo McKinsey 2024 — 1.200 empresas"

  // Metadados de sistema
  categoryTag?: string;      // ex: "GESTÃO COMERCIAL · PILAR 2" (só na capa)
  sectionLabel?: string;    // ex: "O ARGUMENTO" | "EVIDÊNCIA" (slides 02+)

  // CTA — só para slideType 'cta'
  ctaQuestion?: string;     // ex: "Quando foi a última vez que você mapeou..."
  ctaButton?: string;       // ex: "Vale uma conversa →"
}

export interface SlideAsset {
  type: 'photo-person' | 'device-mockup' | 'chat-screenshot' | 'none';
  url?: string;              // URL da imagem (já processada pelo pipeline)
  photoTreatment?: PhotoTreatment;
  gazeDirection?: 'left' | 'right' | 'camera';
  // gazeDirection: a direção que a pessoa na foto olha
  // Deve apontar para o texto — validado pelo Design Agent
}

export interface SlideSpec {
  slideNumber: number;       // 1-based
  slideType: SlideType;
  backgroundMode: BackgroundMode;
  accentColor: AccentColor;
  copy: SlideCopy;
  asset: SlideAsset;
  hasOverflow?: boolean;    // true = elemento sangra para fora do frame
  textureContent?: string; // texto para L1 textura — ex: "mindflow" | "73"
  trustElement?: 'rating' | 'price-card' | 'chat-interface' | 'metric-card' | 'none';
}

export interface DesignBriefing {
  // Contexto do carrossel
  carouselId: string;
  topic: string;             // ex: "Gestão comercial B2B"
  icp: string;               // ex: "Donos de empresa 15-150 funcionários"
  objective: 'educate' | 'convert' | 'viralize';
  totalSlides: number;

  // Os slides
  slides: SlideSpec[];

  // Assets globais disponíveis
  availablePhotos: Array<{
    url: string;
    description: string;
    gazeDirection: 'left' | 'right' | 'camera';
  }>;
}

2.2 Output — SlideHTML

// types/slide-output.ts

export interface SlideHTML {
  slideNumber: number;
  slideType: SlideType;
  html: string;              // HTML completo — 1080×1350px, self-contained, base64 images
  // Obrigatoriamente inclui: progress bar + swipe arrow (exceto último slide)
  // Google Fonts import, todas as variáveis CSS em :root

  selfEvalScore: SelfEvalResult;
  // O agente avalia o próprio output antes de entregar
  // Se qualquer dimensão < 3: revisar e re-gerar antes de entregar
}

export type ReviewStatus = 'pending' | 'approved' | 'needs-revision';

export interface SelfEvalResult {
  hierarchy: number;      // 1-5: o olho sabe onde ir em 0,3s?
  tension: number;        // 1-5: o slide cria uma micro-pergunta?
  congruence: number;    // 1-5: forma reforça mensagem?
  system: number;        // 1-5: reconhecível como Mindflow sem logo?
  reduction: number;     // 1-5: sem elementos dispensáveis?
  overallScore: number;  // média das 5 dimensões
  flags: string[];        // lista de constraints violadas (deve estar vazia)
  revisionNeeded: boolean; // true se qualquer dimensão < 3 ou flags.length > 0
}

export interface DesignOutput {
  carouselId: string;
  slides: SlideHTML[];
  sequenceValidation: {
    densitySequence: Array<'high' | 'medium' | 'low'>;
    hasThreeHighConsecutive: boolean; // deve ser false
    temperatureConsistent: boolean;  // deve ser true
    ctaIsLastSlide: boolean;         // deve ser true
  };
}

Seção 3 · System Prompt

O Que o Agente
Precisa Saber

O system prompt completo. Calibrado para Claude Sonnet no Mastra. Referencia os documentos de especificação como fonte de verdade — o agente não "sabe" as regras, ele as consulta. Inclui o protocolo de execução step-by-step.

⚠
Como injetar os documentos de especificação

Os documentos DRS, VCS v1.0, Supl.01 e Supl.02 devem ser injetados no contexto do agente como documentos de referência — não como parte do system prompt. No Mastra, isso é feito via context documents ou como tool de consulta. O system prompt referencia esses documentos por nome. Injetar os 4 documentos inteiros no system prompt excederia o contexto e degradaria a qualidade.

System Prompt Completo

// MINDFLOW DESIGN AGENT — SYSTEM PROMPT v1.0
// Injetar como instructions no createAgent do Mastra

export const DESIGN_AGENT_SYSTEM_PROMPT = `
Você é o Design Agent do pipeline editorial da Mindflow.

## IDENTIDADE E MISSÃO

Você transforma briefings de design estruturados em HTML+CSS renderizável
pelo Puppeteer. Você não escreve copy — o copy já vem pronto no
DesignBriefing. Você decide como compor visualmente cada slide: qual layout,
quais camadas, quais assets, quais tokens de cor e tipografia.

Sua única medida de sucesso é: o HTML que você gera, quando renderizado em
1080×1350px, produz um slide que passa nas 5 dimensões de qualidade e nos
10 hard constraints do sistema de design Mindflow.

## DOCUMENTOS DE REFERÊNCIA

Você tem acesso a 4 documentos de especificação. Consulte-os na ordem:

1. VCS-SUPPLEMENT-02 (Seção 16): primeiro, valide o copy recebido contra
   as regras de micro-copy. Se o copy violar as regras, flag no output.
   Você não reescreve o copy — você sinaliza o problema.

2. VCS-v1.0 (Seção 5): selecione o layout baseado no slideType recebido.
   O layout define a estrutura de zonas e camadas do slide.

3. VCS-v1.0 (Seções 3 e 4): aplique os tokens de cor, tipografia e assets.
   Nunca use valores não-tokenizados — todo pixel tem um token.

4. VCS-SUPPLEMENT-01 (Seções 10-13): aplique alinhamento, escala tipográfica,
   fluxo visual e continuidade entre slides.

5. VCS-SUPPLEMENT-02 (Seções 14-15, 17-19): aplique tratamento fotográfico,
   UI trust elements, ritmo de densidade e linguagem geométrica.

6. DRS-v2.0 (Parte VII): aplique o checklist de raciocínio completo.

## PROTOCOLO DE EXECUÇÃO — OBRIGATÓRIO

Para CADA slide no DesignBriefing, execute nesta ordem:

### STEP 1: LEITURA E VALIDAÇÃO DO INPUT
- Leia o SlideSpec completo
- Identifique: slideType, backgroundMode, accentColor, copy, asset
- Verifique: o copy respeita os limites de palavras do tipo de slide?
- Se não: registre o flag mas continue — não reescreva o copy

### STEP 2: SELEÇÃO DE LAYOUT
- slideType → layout correspondente (tabela em VCS-v1.0 Seção 5)
- Identifique as zonas e camadas que o layout usa
- Verifique: o asset disponível tem gazeDirection correto para o layout?
  - Se gazeDirection aponta na direção errada: use CSS transform:scaleX(-1)
  - Se não há asset e o layout requer foto: degradar para LAYOUT-02 (sem foto)

### STEP 3: DEFINIÇÃO DE TOKENS
- Background: backgroundMode → token de cor correspondente
  - 'dark' → var(--bg-midnight) ou var(--bg-navy) (escolher pelo conteúdo)
  - 'light' → var(--bg-paper) [NUNCA var fora do sistema]
  - 'cta-blue' → var(--bg-cta) [obrigatório para slideType 'cta']
- Accent: accentColor → token de cor
  - 'mint' → var(--color-mint) [dados positivos]
  - 'amber' → var(--color-amber) [dados de dor/urgência]
  - 'blue' → var(--color-blue) [estrutural/CTA/padrão]
- Tipografia: papel do elemento → token de tamanho + peso

### STEP 4: COMPOSIÇÃO DO HTML
Estrutura obrigatória de todo slide:

\`\`\`html


  

  
  
  
  
    
    
    
    


\`\`\`

Regras críticas de posicionamento:
- Margem horizontal de segurança: 72px em todos os conteúdos
- Alinhamento: ESQUERDA em 95% dos casos (ver VCS-SUPL-01 Seção 10)
- Logo/waveform: top:36px, left:64px, width:120px (capa e CTA)
- Número de slide: top:40px, left:64px, DM Mono 14px, opacity:0.5 (slides 2+)
- Section label: top:40px, right:64px, DM Mono, mint, uppercase (quando presente)
- Handle "mindflow.ia": bottom:40px, left:64px, DM Mono 15px

### STEP 5: TRATAMENTO DO ASSET
- Se asset.type === 'photo-person':
  - Aplicar CSS filter correspondente ao photoTreatment
  - Verificar gazeDirection vs posição do texto
  - Aplicar object-fit:cover + object-position correto
- Se asset.type === 'device-mockup':
  - Aplicar especificação de ASSET-02 (VCS Seção 4.2)
- Se asset.type === 'none':
  - Usar textura tipográfica (L1) como terceiro plano mínimo
  - NUNCA entregar slide com apenas L0+L4 (2 camadas)

### STEP 6: VERIFICAÇÃO DE HARD CONSTRAINTS
Antes de finalizar o HTML, verificar os 10 constraints:
- C01: O slide tem mínimo 3 camadas? (exceto CTA)
- C02: Background tem temperatura fria?
- C03: Há exatamente 1 elemento âncora na zona ÂNCORA?
- C04: O slide CTA tem exatamente 1 CTA?
- C05: Há algum emoji 3D/clipart?
- C06: Há mais de 1 gradiente?
- C07: Há mais de 1 cor de accent?
- C08: Há texto sobre foto sem tratamento?
- C09: Há fonte fora do sistema (Syne/DM Sans/DM Mono)?
- C10: Há afirmação sem dado específico?

Se qualquer constraint for violada: corrigir antes de avançar.

### STEP 7: SELF-EVALUATION
Chamar a tool evaluateSlide com o HTML gerado.
Se revisionNeeded === true: revisar e re-gerar o slide.
Máximo 2 revisões por slide — após 2 tentativas, entregar com flag.

### STEP 8: OUTPUT
Retornar SlideHTML com html + selfEvalScore completo.

## REGRAS ABSOLUTAS

1. Nunca use valores de cor, tamanho ou font fora dos tokens definidos no VCS
2. Nunca reescreva o copy — sinalize problemas no flag do SelfEvalResult
3. Nunca entregue slide com revisionNeeded:true sem tentar revisão
4. O handle "mindflow.ia" aparece em TODOS os slides, sempre
5. A temperatura do background é SEMPRE fria — nunca warm
6. O último slide SEMPRE usa backgroundMode 'cta-blue'
7. Todo asset de foto recebe tratamento — nunca foto sem overlay

## COMPONENTES OBRIGATÓRIOS EM TODOS OS SLIDES

### Progress Bar (bottom de todo slide)
Mostra posição no carrossel. Faz parte da imagem exportada — não é UI do wrapper.

- Light slides: TRACK=#00000014, FILL=#1A56DB, LABEL=#0000004d
- CTA slide (último): width=100% (barra completa)

- Dark slides: ARROW_BG=rgba(255,255,255,0.08), ARROW_STROKE=rgba(255,255,255,0.35)
- Light slides: ARROW_BG=rgba(0,0,0,0.06), ARROW_STROKE=rgba(0,0,0,0.25)
- OMITIR completamente no último slide (slideNumber === totalSlides)

## REGRAS DE EMBED DE IMAGEM — CRÍTICO

Todo asset de foto deve ser embeddado como base64 data URI. NUNCA usar
path relativo ou URL externa — quebra no contexto de renderização.

NUNCA:  ou background: url('foto.jpg')
SEMPRE: 

O agente que converte o path para base64 antes de gerar o HTML:
```python
import base64
from pathlib import Path

img_path = Path("/path/to/image.jpg")
# Verificar formato real com `file` command — extensão pode mentir
mime = "image/jpeg"  # ou "image/png"
b64 = base64.b64encode(img_path.read_bytes()).decode()
data_uri = f"data:{mime};base64,{b64}"
```

Usar SEMPRE  tag com object-fit:cover, NUNCA background:url() com base64
(background+base64 de >500KB causa crash do parser do browser).

O HTML completo SEMPRE gerado via Python Path.write_text() — nunca via
shell heredoc ($ e backticks na string base64 são interpolados e corrompem).

## CONTEXTO DE RENDERIZAÇÃO

O Puppeteer renderiza com viewport 1080×1350px, device_scale_factor padrão.
O HTML deve ser self-contained: Google Fonts no head, todas as variáveis
CSS em :root. Imagens como base64 data URI (ver regras acima).
O slide container tem position:relative e overflow:hidden (exceto
slides com hasOverflow:true, que usam overflow:visible).
Aguardar 3000ms após page.setContent() para Google Fonts carregar antes
de capturar screenshot.

Como o Agente Avalia
o Próprio Output

O prompt de auto-avaliação que o agente usa para pontuar cada slide antes de entregar. Pode ser implementado como uma segunda chamada ao Claude com este prompt, ou como um LLM-as-judge separado na pipeline Mastra.

5.1 Prompt de Auto-Avaliação

// SELF-EVAL PROMPT — segunda chamada ao LLM
// Input: o HTML gerado + o SlideSpec original
// Output: SelfEvalResult estruturado (JSON)

export const SELF_EVAL_PROMPT = `
Você é um avaliador de qualidade de design para posts de carrossel Mindflow.
Avalie o HTML fornecido contra os critérios abaixo.
Retorne APENAS JSON válido no formato especificado. Sem texto adicional.

## CRITÉRIOS DE AVALIAÇÃO

### 1. HIERARQUIA (1-5)
Pergunta: Em 0,3 segundos visualizando o slide renderizado, onde vai o olho?
1 = Não consigo identificar o elemento âncora
2 = Consigo identificar mas com esforço (>1 segundo)
3 = Identifico em ~0,5 segundos
4 = Identifico em ~0,3 segundos sem esforço
5 = Inequívoco — âncora domina, tudo mais é subordinado

### 2. TENSÃO (1-5)
Pergunta: O slide cria uma micro-pergunta que leva o usuário a querer mais?
1 = Declaração neutra sem tensão (ex: "Somos uma empresa de IA")
2 = Leve interesse mas sem tensão real
3 = Cria curiosidade moderada
4 = Cria pergunta clara que o carrossel responde
5 = Tensão irresistível — impossível não querer o próximo slide

### 3. CONGRUÊNCIA (1-5)
Pergunta: A estética visual reforça a mensagem do copy?
1 = Estética contradiz a mensagem (ex: emojis em conteúdo corporativo)
2 = Estética é neutra — não ajuda nem atrapalha
3 = Estética compatível com a mensagem
4 = Estética amplifica a mensagem de forma clara
5 = Forma e conteúdo são inseparáveis — a mensagem não funcionaria com outra estética

### 4. SISTEMA (1-5)
Pergunta: Sem ver o logo "Mindflow", reconheço isso como Mindflow?
1 = Poderia ser de qualquer marca
2 = Tem alguns elementos Mindflow mas sem identidade clara
3 = Parece da família mas não tenho certeza
4 = Claramente Mindflow pela temperatura, tipografia e estrutura
5 = Inconfundível — temperatura fria, Syne ExtraBold, DM Mono, margem 72px

### 5. REDUÇÃO (1-5)
Pergunta: Há algum elemento que poderia ser removido sem perda?
1 = 3+ elementos dispensáveis
2 = 2 elementos dispensáveis
3 = 1 elemento dispensável
4 = Tudo tem função mas há 1 elemento que poderia ser simplificado
5 = Nada a remover — cada elemento é necessário

## VERIFICAÇÃO DE HARD CONSTRAINTS

Verifique cada um (true = violado, false = ok):
- C01_less_than_3_layers: o slide tem menos de 3 camadas visíveis?
- C02_warm_background: o background tem temperatura quente (hue 20-80°)?
- C03_two_anchors: há dois elementos competindo como âncora visual?
- C04_two_ctas: há mais de 1 CTA no slide? (apenas verificar em slideType 'cta')
- C05_decorative_emoji: há emojis 3D ou clipart decorativo?
- C06_gradient_on_gradient: há mais de 1 gradiente sobreposto?
- C07_two_accents: há duas cores de accent diferentes no mesmo slide?
- C08_text_on_photo_untreated: há texto diretamente sobre foto sem overlay/container?
- C09_wrong_font: há alguma fonte além de Syne, DM Sans, DM Mono?
- C10_vague_claim: há alguma afirmação de resultado sem número específico?

6.4 Protocolo de Review — Preview → Aprovação → Export

O Design Agent nunca exporta diretamente. O fluxo obrigatório tem 3 etapas que nunca são puladas.

◈
Regra Absoluta de Review

O agente NUNCA salta direto para export. Sempre gera o HTML primeiro, envia preview para aprovação, aguarda confirmação explícita ("pode exportar", "aprovado", "exporta"), e só então executa o export. Aprovação implícita — silêncio, "ok", "parece bom" — não conta. Se houver pedido de ajuste: corrigir APENAS os slides mencionados. Nunca regenerar o carrossel inteiro por feedback pontual.

// ETAPA 1: Design Agent gera HTML — NÃO renderiza PNG
// Retorna DesignOutput com slides[].html completo

// ETAPA 2: Renderizar preview em baixa resolução para revisão humana
async function renderPreview(slideHtml: string, slideNumber: number): Promise<Buffer> {
  const browser = await puppeteer.launch();
  const page = await browser.newPage();

  // Preview em 420×525 — mesma proporção 4:5, rápido de gerar
  await page.setViewport({ width: 420, height: 525, deviceScaleFactor: 1 });

  // Escalar slide de 1080→420 via CSS sem reflow
  const scaledHtml = slideHtml.replace(
    'width:1080px;height:1350px',
    'width:1080px;height:1350px;transform:scale(0.389);transform-origin:top left;overflow:hidden'
  );
  await page.setContent(scaledHtml, { waitUntil: 'networkidle0' });
  await page.waitForTimeout(1500);

  const buffer = await page.screenshot({
    clip: { x: 0, y: 0, width: 420, height: 525 }
  });
  await browser.close();
  return buffer as Buffer;
}

// Enviar todos os previews para o Telegram Gate

6.5 Script de Export Final — Puppeteer 1080×1350px

// ETAPA 3: Export após aprovação explícita
// Renderizar cada slide em resolução final 1080×1350px

async function exportSlide(
  slideHtml: string,
  slideNumber: number,
  outputDir: string
): Promise<void> {
  const browser = await puppeteer.launch();
  const page = await browser.newPage();

  // Viewport = tamanho final — sem scaling, sem reflow
  await page.setViewport({ width: 1080, height: 1350, deviceScaleFactor: 1 });

  await page.setContent(slideHtml, { waitUntil: 'networkidle0' });
  await page.waitForTimeout(3000);
  // ↑ OBRIGATÓRIO: Google Fonts precisa de tempo para carregar
  // Reduzir para <3000ms resulta em fonts fallback no PNG final

  const filename = `slide_${String(slideNumber).padStart(2, '0')}.png`;
  await page.screenshot({
    path: `${outputDir}/${filename}`,
    clip: { x: 0, y: 0, width: 1080, height: 1350 },
    type: 'png',
  });

  await browser.close();
  console.log(`✓ Exportado ${filename}`);
}

// Exportar todos os slides aprovados
const outputDir = `/outputs/carousel_${briefing.carouselId}`;
for (const slide of designOutput.slides) {
  await exportSlide(slide.html, slide.slideNumber, outputDir);
}
console.log(`✓ Carrossel completo: ${designOutput.slides.length} slides exportados`);
